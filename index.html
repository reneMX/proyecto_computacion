Ejemplo de Código Python

# Calculadora de IMC para nutrición
def calcular_imc(peso, altura):
    """Calcula el Índice de Masa Corporal"""
    return peso / (altura ** 2)

def clasificar_imc(imc):
    """Clasifica el IMC en categorías"""
    if imc < 18.5:
        return "Bajo peso"
    elif imc < 25:
        return "Peso normal"
    elif imc < 30:
        return "Sobrepeso"
    else:
        return "Obesidad"

# Ejemplo de uso
peso = 68  # kg
altura = 1.75  # metros
imc = calcular_imc(peso, altura)
categoria = clasificar_imc(imc)
print(f"IMC: {imc:.2f} - Categoría: {categoria}")
